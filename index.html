bubble_sort.cpp 
#include <iostream>
using namespace std;

void bubbleSort(int arr[], int n) {
  for (int i = 0; i < n-1; i++) {
    for (int j = 0; j < n-i-1; j++) {
      if (arr[j] > arr[j+1]) {
        swap(arr[j], arr[j+1]);
      }
    }
  }
}

int main() {
  int arr[] = {64, 34, 25, 12, 22};
  int n = sizeof(arr)/sizeof(arr[0]);
  bubbleSort(arr, n);
  // Output: 12 22 25 34 64
  return 0;
}
